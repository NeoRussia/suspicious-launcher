+// SUSY: thin-film interference for surfaces marked prismatic in their specular map.
+// A thin transparent film over a surface reflects each wavelength with a different phase, so
+// the reflection takes on a hue that depends on the angle between view and surface - the
+// soap-bubble / oil-slick effect. Film thickness drifts slowly with time and varies across
+// world position, so neighbouring faces do not show the same rings.
+vec3 prismaticFilmTint(float NdotV, vec3 worldPos){
+	const float filmIOR = 1.40;
+	const vec3 wavelength = vec3(650.0, 532.0, 460.0); // nm, R G B
+
+	float thickness = 380.0
+		+ 110.0 * sin(TIME * 0.35 + dot(worldPos, vec3(1.7, 2.3, 1.3)))
+		+ 70.0  * sin(TIME * 0.61 - dot(worldPos, vec3(2.1, 1.1, 2.9)));
+

+	float cosT = sqrt(max(1.0 - (1.0 - NdotV*NdotV) / (filmIOR*filmIOR), 0.0));
